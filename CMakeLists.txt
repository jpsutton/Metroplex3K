cmake_minimum_required(VERSION 3.16)
project(Metroplex3K LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# Platform-independent engine core: no SDL, no OS APIs beyond the C++ standard library.
file(GLOB CORE_SRC CONFIGURE_DEPENDS src/core/*.cpp)
add_library(m3k_core STATIC ${CORE_SRC})
target_include_directories(m3k_core PUBLIC src)
find_package(Threads REQUIRED)  # audio decode and render worker threads
target_link_libraries(m3k_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(m3k_core PRIVATE /W4)
else()
  target_compile_options(m3k_core PRIVATE -Wall -Wextra)
endif()

# Tests: plain executables, no framework dependency.
enable_testing()
file(GLOB TEST_SRC CONFIGURE_DEPENDS tests/test_*.cpp)
foreach(t ${TEST_SRC})
  get_filename_component(name ${t} NAME_WE)
  add_executable(${name} ${t})
  target_link_libraries(${name} PRIVATE m3k_core)
  add_test(NAME ${name} COMMAND ${name})
endforeach()

# Developer tools.
file(GLOB TOOL_SRC CONFIGURE_DEPENDS tools/*.cpp)
foreach(t ${TOOL_SRC})
  get_filename_component(name ${t} NAME_WE)
  # render_parity needs SDL2 + OpenGL compute; built with the app stack below.
  if(NOT name STREQUAL "render_parity" AND NOT name STREQUAL "gl_present_check")
    add_executable(${name} ${t})
    target_link_libraries(${name} PRIVATE m3k_core)
  endif()
endforeach()

# Probes: engine-vs-real-install checks (tools/probes/README.md). Not tests: CI has no game install.
option(M3K_BUILD_PROBES "Build the real-install probes in tools/probes" ON)
if(M3K_BUILD_PROBES)
  file(GLOB PROBE_SRC CONFIGURE_DEPENDS tools/probes/probe_*.cpp)
  foreach(t ${PROBE_SRC})
    get_filename_component(name ${t} NAME_WE)
    add_executable(${name} ${t})
    target_include_directories(${name} PRIVATE tools/probes)
    target_link_libraries(${name} PRIVATE m3k_core)
  endforeach()
endif()

# Game application (SDL2 desktop host). M3K_BUILD_APP=OFF builds core, tests and tools only (no SDL needed).
# With the option ON and SDL2 missing, the app is skipped with a status message so core + tests still build.
option(M3K_BUILD_APP "Build the SDL2 desktop application (src/app)" ON)
if(M3K_BUILD_APP)
  find_package(SDL2 QUIET)
  if(SDL2_FOUND)
    file(GLOB APP_SRC CONFIGURE_DEPENDS src/app/*.cpp)
    add_executable(metroplex3k ${APP_SRC})
    target_link_libraries(metroplex3k PRIVATE m3k_core)
    if(TARGET SDL2::SDL2main AND WIN32)
      target_link_libraries(metroplex3k PRIVATE SDL2::SDL2main)
    endif()
    if(TARGET SDL2::SDL2)
      target_link_libraries(metroplex3k PRIVATE SDL2::SDL2)
    else()  # older SDL2 packages only export variables
      target_include_directories(metroplex3k PRIVATE ${SDL2_INCLUDE_DIRS})
      target_link_libraries(metroplex3k PRIVATE ${SDL2_LIBRARIES})
    endif()
    # Phase 2 terrain compute (optional): OpenGL 4.3 + GLEW. App still builds without them; GPU path fails closed.
    find_package(OpenGL QUIET)
    find_path(GLEW_INCLUDE_DIR GL/glew.h)
    find_library(GLEW_LIBRARY GLEW)
    if(OpenGL_FOUND AND GLEW_INCLUDE_DIR AND GLEW_LIBRARY)
      target_include_directories(metroplex3k PRIVATE ${GLEW_INCLUDE_DIR})
      target_link_libraries(metroplex3k PRIVATE OpenGL::GL ${GLEW_LIBRARY})
      target_compile_definitions(metroplex3k PRIVATE M3K_HAS_GL_COMPUTE=1)

      add_executable(render_parity tools/render_parity.cpp src/app/gl_compute.cpp)
      target_include_directories(render_parity PRIVATE src src/app ${GLEW_INCLUDE_DIR})
      target_link_libraries(render_parity PRIVATE m3k_core OpenGL::GL ${GLEW_LIBRARY})
      target_compile_definitions(render_parity PRIVATE M3K_HAS_GL_COMPUTE=1)
      target_compile_definitions(metroplex3k PRIVATE M3K_HAS_GL_PRESENT=1)
      add_executable(gl_present_check tools/gl_present_check.cpp src/app/gl_presenter.cpp)
      target_compile_definitions(gl_present_check PRIVATE M3K_HAS_GL_PRESENT=1)
      target_include_directories(gl_present_check PRIVATE src src/app ${GLEW_INCLUDE_DIR})
      target_link_libraries(gl_present_check PRIVATE m3k_core OpenGL::GL ${GLEW_LIBRARY})
      if(TARGET SDL2::SDL2)
        target_link_libraries(gl_present_check PRIVATE SDL2::SDL2)
      else()
        target_include_directories(gl_present_check PRIVATE ${SDL2_INCLUDE_DIRS})
        target_link_libraries(gl_present_check PRIVATE ${SDL2_LIBRARIES})
      endif()
      if(TARGET SDL2::SDL2)
        target_link_libraries(render_parity PRIVATE SDL2::SDL2)
      else()
        target_include_directories(render_parity PRIVATE ${SDL2_INCLUDE_DIRS})
        target_link_libraries(render_parity PRIVATE ${SDL2_LIBRARIES})
      endif()
    else()
      message(STATUS "GLEW/OpenGL not found: metroplex3k builds without GPU terrain compute")
      target_compile_definitions(metroplex3k PRIVATE M3K_HAS_GL_COMPUTE=0)
    endif()
  else()
    message(STATUS "SDL2 not found: building core, tests and tools only (install SDL2 to build the app)")
  endif()
else()
  message(STATUS "M3K_BUILD_APP=OFF: building core, tests and tools only")
endif()
